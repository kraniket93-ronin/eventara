# Eventara - https://www.eventara.co.in
# Public marketplace pages are open to crawlers. Account, transaction and
# internal pages are excluded: they need a session, so a crawler would only
# ever see an empty shell or a redirect, and indexing them would spend crawl
# budget on pages that can never rank.

User-agent: *
Allow: /

# Signed-in and transactional areas
Disallow: /customer-dashboard.html
Disallow: /supplier-dashboard.html
Disallow: /ops.html
Disallow: /signin.html
Disallow: /verified.html
Disallow: /booking.html
Disallow: /invoice.html
Disallow: /compare.html

# Legacy stub - vercel.json 301s this to /venue/paandora-grand-udaipur
Disallow: /provider.html

# NOTE: /supplier.html?slug=... is deliberately NOT disallowed. It renders the
# same supplier as /venue/<slug> and carries a canonical pointing there, so
# Google needs to crawl it to see that canonical and consolidate the two.
# Blocking it would leave the duplicate unresolved.

# Auth callbacks carry one-time tokens in the query string
Disallow: /*?recovery=
Disallow: /*?verified=

Sitemap: https://www.eventara.co.in/sitemap.xml
